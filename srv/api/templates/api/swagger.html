Python client

Prefer ACOBClient over raw HTTP. Install from the monorepo (python -m pip install ./client) and target this stack explicitly; await client.api() returns the same links and guide as GET /api/ without browser work.

from acob import ACOBClient

async with ACOBClient(endpoint="{{ docs.base_url }}") as client:
    tabs = await client.list()
    tab = await client.navigate("https://example.com")
    screenshot = await client.screenshot(tab.tid)
    print(screenshot.url)
